#4. Write a Python Program to Find the Real & Complex Eigen Values and Eigen Vectors of a #User-defined Matrix with Visualization import numpy as np import cmath import matplotlib.pyplot as plt n = int(input("Enter the order of the matrix")) A = [] for i in range(n): row = list(map(float, input().split())) A.append(row) print("A=", A) e, V = np.linalg.eig(A) print("Eigen values of A: ", e) print("Eigen vector of A(column-wise): ", V) for i in range(n): v = V[:, i] Av = A @ v print("Av=", Av) plt.quiver(0, 0, v[0], v[1], angles='xy', scale_units='xy', scale=1, label="x", color='blue') plt.quiver(0, 0, -v[0], -v[1], angles='xy', scale_units='xy', scale=1, label="x", color='blue') plt.quiver(0, 0, Av[0], Av[1], angles='xy', scale_units='xy', scale=1, label="Ax", color='red') plt.quiver(0, 0, -Av[0], -Av[1], angles='xy', scale_units='xy', scale=1, label="Ax", color='red') plt.xlim(-6, 6) plt.ylim(-6, 6) plt.axhline(0, color="black") plt.axvline(0, color="black") plt.title("Visualization of Eigenvectors of the Given Matrix") plt.grid(True) plt.legend() plt.show()

#2. Write a Python Program to find the root of a User-defined Transcendental Equation with #Visualization using Newton Raphson Method import numpy as np import matplotlib.pyplot as plt x = float(input("Enter the initial guess x0: ")) function = input("Enter the function (use x as the variable): ") derivative = input("Enter the derivative of the function: ") n = abs(int(input("Enter the number of iterations: "))) def f(x): return eval(function) def df(x): return eval(derivative) print("-" * 120) print(f"{'S.NO':<6}|{'x':<16}|{'f(x)':<20}|{'f\'(x)':<20}|{'x_new':<16}") print("-" * 120) for i in range(n): fx = f(x) dfx = df(x) if dfx == 0: print("Derivative is zero. Newton-Raphson method fails.") break x_new = x - fx / dfx print(f"{i+1:<6}|{x:<16.6f}|{fx:<20.6f}|{dfx:<20.6f}|{x_new:<16.6f}") print("-" * 120) x = x_new print("Approximate root =", x) # Plot x = np.linspace(1, 3, 400) y = f(x) plt.plot(x, y, label=r"$f(x)= x*\log(x) - 1.2$") plt.axhline(0, color="black") plt.grid(True) plt.legend() plt.xlabel("x") plt.ylabel("f(x)") plt.title("Function graph") plt.show()

#1. Write a Python program to solve the following natural growth problem: The number N of #bacteria in a culture grows at a rate proportional to N. The value of N was initially 100 #and increased to 332 in one hour. What was the value of N after 1/2 hour? import numpy as np import matplotlib.pyplot as plt # User-defined function to calculate growth/decay constant def find_constant(P0, t1, P1): k = (1 / t1) * np.log(P1 / P0) return k # User-defined function to find quantity def find_quantity(P0, k, t): P = P0 * np.exp(k * t) return P # User-defined function to find time def find_time(P0, k, P): t = (1 / k) * np.log(P / P0) return t # Input data P0 = float(input("Enter initial quantity P0 = ")) t1 = float(input("Enter known time t1 = ")) P1 = float(input("Enter quantity at t1 = ")) # Calculate constant k = find_constant(P0, t1, P1) print("Growth/Decay constant k =", k) # Check growth or decay if k > 0: print("The quantity represents Natural Growth.") elif k < 0: print("The quantity represents Natural Decay.") else: print("The quantity remains constant.") # Find P2 or t2 print("For P2 and t2:") print("1. Find P2 for a given t2") print("2. Find t2 for a given P2") print("3. Skip P2 and t2") choice = int(input("Enter your choice = ")) P2 = None t2 = None if choice == 1: t2 = float(input("Enter time t2 = ")) P2 = find_quantity(P0, k, t2) print("Time t2 =", t2) print("Quantity P2 =", P2) elif choice == 2: P2 = float(input("Enter required quantity P2 = ")) t2 = find_time(P0, k, P2) print("Quantity P2 =", P2) print("Time t2 =", t2) elif choice == 3: print("P2 and t2 calculation skipped.") else: print("Invalid choice") # Visualization if choice in (1, 2): t = np.linspace(0, max(t1, t2) + 1, 100) P = P0 * np.exp(k * t) plt.plot(t, P) plt.scatter([t1, t2], [P1, P2]) plt.xlabel("Time") plt.ylabel("Quantity") plt.title("Law of Natural Growth and Decay") plt.grid(True) plt.show()
